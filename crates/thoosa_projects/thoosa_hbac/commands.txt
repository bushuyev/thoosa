use interprocess::local_socket::{traits::tokio::{Listener as _, Stream as _}, GenericFilePath, GenericNamespaced, ListenerOptions, NameType, ToFsName, ToNsName};
use interprocess::local_socket::tokio::Stream;
use serde::{Deserialize, Serialize};
use std::error::Error;
use tokio::io::{AsyncBufReadExt, AsyncWrite, AsyncWriteExt, BufReader};
use tokio::sync::{mpsc, watch};
use tokio::time::{sleep, Duration};

type Result<T> = std::result::Result<T, Box<dyn Error + Send + Sync>>;

#[derive(Debug, Clone, Serialize, Deserialize)]
struct CollectEtaInvocation {
    stream_id: u32,
    completed: u64,
    total: u64,
}

#[derive(Debug, Clone, Default, Serialize, Deserialize)]
struct CollectEtaState {
    invocations: Vec<CollectEtaInvocation>,
}

#[derive(Debug, Clone)]
struct ProgressUpdate {
    stream_id: u32,
    current: u64,
    total: u64,
}

#[tokio::main]
async fn main() -> Result<()> {
    let mode = std::env::args().nth(1).unwrap_or_else(|| "server".to_string());

    match mode.as_str() {
        "server" => run_server().await,
        "client" => run_client_forever().await,
        other => Err(format!("unknown mode: {other} (use: server | client)").into()),
    }
}

#[tokio::test(flavor = "multi_thread", worker_threads = 16)]
async fn test_server() -> Result<()>{
    run_server().await?;

    Ok(())
}

#[tokio::test(flavor = "multi_thread", worker_threads = 16)]
async fn test_client() -> Result<()>{
    run_client_once().await?;

    Ok(())
}


/// Cross-platform-ish socket name helper.
/// - Prefer a namespaced local socket when supported.
/// - Otherwise fall back to a filesystem path on Unix or a named-pipe path on Windows.
fn socket_name() -> Result<interprocess::local_socket::Name<'static>> {
    if GenericNamespaced::is_supported() {
        Ok("thoosa-progress".to_ns_name::<GenericNamespaced>()?)
    } else if cfg!(windows) {
        Ok(r"\\.\pipe\thoosa-progress".to_fs_name::<GenericFilePath>()?)
    } else {
        Ok("/tmp/thoosa-progress.sock".to_fs_name::<GenericFilePath>()?)
    }
}

async fn run_server() -> Result<()> {
    let name = socket_name()?;
    let listener = ListenerOptions::new()
        .name(name)
        .create_tokio()?;

    // Workers send point updates here.
    let (updates_tx, mut updates_rx) = mpsc::unbounded_channel::<ProgressUpdate>();

    // Every connected UI gets snapshots from here.
    let (snapshot_tx, snapshot_rx) = watch::channel(CollectEtaState::default());

    // Aggregator task: own the latest snapshot, apply updates, publish new snapshots.
    tokio::spawn(async move {
        let mut snapshot = CollectEtaState::default();

        while let Some(update) = updates_rx.recv().await {
            apply_update(&mut snapshot, update);//for demo purposes, to see updates
            let _ = snapshot_tx.send(snapshot.clone());
        }
    });

    // Demo workers. Replace these with your real parallel streams/tasks.
    for stream_id in 0..4u32 {
        let tx = updates_tx.clone();
        tokio::spawn(async move {
            fake_worker(stream_id, 100, tx).await;
        });
    }
    drop(updates_tx);

    println!("server listening");

    loop {
        let stream = listener.accept().await?;
        let rx = snapshot_rx.clone();

        tokio::spawn(async move {
            if let Err(e) = serve_one_client(stream, rx).await {
                eprintln!("client connection ended: {e}");
            }
        });
    }
}

async fn serve_one_client(
    stream: Stream,
    mut rx: watch::Receiver<CollectEtaState>,
) -> Result<()> {
    let (_reader, mut writer) = stream.split();

    // Immediately send latest known state to a newly connected UI.
    let first = rx.borrow().clone();
    send_snapshot(&mut writer, &first).await?;

    // Then push every new snapshot.
    loop {
        if rx.changed().await.is_err() {
            break;
        }

        let snapshot = rx.borrow_and_update().clone();
        send_snapshot(&mut writer, &snapshot).await?;
    }

    Ok(())
}

async fn send_snapshot<W>(writer: &mut W, snapshot: &CollectEtaState) -> Result<()>
where
    W: AsyncWrite + Unpin,
{
    let mut bytes = serde_json::to_vec(snapshot)?;
    bytes.push(b'\n');
    writer.write_all(&bytes).await?;
    writer.flush().await?;
    Ok(())
}

fn apply_update(snapshot: &mut CollectEtaState, update: ProgressUpdate) {
    if let Some(s) = snapshot
        .invocations
        .iter_mut()
        .find(|s| s.stream_id == update.stream_id)
    {
        s.completed = update.current;
        s.total = update.total;
    } else {
        snapshot.invocations.push(CollectEtaInvocation {
            stream_id: update.stream_id,
            completed: update.current,
            total: update.total,
        });
        snapshot.invocations.sort_by_key(|s| s.stream_id);
    }
}

async fn fake_worker(
    stream_id: u32,
    total: u64,
    updates_tx: mpsc::UnboundedSender<ProgressUpdate>,
) {
    for current in 0..=total {
        let _ = updates_tx.send(ProgressUpdate {
            stream_id,
            current,
            total,
        });

        let delay_ms = 30 + (stream_id as u64) * 20;
        sleep(Duration::from_millis(delay_ms)).await;
    }
}

async fn run_client_forever() -> Result<()> {
    loop {
        match run_client_once().await {
            Ok(()) => {
                eprintln!("server disconnected, reconnecting...");
            }
            Err(e) => {
                eprintln!("not connected yet: {e}");
            }
        }

        sleep(Duration::from_secs(1)).await;
    }
}

async fn run_client_once() -> Result<()> {
    let name = socket_name()?;
    let stream = Stream::connect(name).await?;
    let (reader, _writer) = stream.split();

    let mut lines = BufReader::new(reader).lines();

    while let Some(line) = lines.next_line().await? {
        let snapshot: CollectEtaState = serde_json::from_str(&line)?;
        render_snapshot(&snapshot);
    }

    Ok(())
}

fn render_snapshot(snapshot: &CollectEtaState) {
    print!("\x1B[2J\x1B[H"); // clear screen for demo
    println!("Latest progress snapshot:");
    for s in &snapshot.invocations {
        println!("stream {:>2}: {:>3} / {}", s.stream_id, s.completed, s.total);
    }
}
